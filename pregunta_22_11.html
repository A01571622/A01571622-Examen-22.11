# 3. Pruebas de estacionariedad
adf_results = []
kpss_results = []

def unit_root_tests(series, name):
    print(f"\nPruebas de Raíz Unitaria para {name}:")
    # Prueba ADF
    adf_result = adfuller(series)
    adf_stat = adf_result[0]
    adf_pvalue = adf_result[1]
    adf_critical = adf_result[4]
    adf_interpret = "Estacionaria" if adf_pvalue < 0.05 else "No estacionaria"
    adf_results.append({
        'Serie': name,
        'Estadístico ADF': f"{adf_stat:.4f}",
        'p-valor ADF': f"{adf_pvalue:.4f}",
        'Valor Crítico 5% ADF': f"{adf_critical['5%']:.4f}",
        'Interpretación ADF': adf_interpret
    })
    print("Prueba ADF:")
    print(f'Estadístico ADF: {adf_stat:.4f}')
    print(f'p-valor: {adf_pvalue:.4f}')
    print(f'Valores Críticos: {adf_critical}')
    print(f"Interpretación: p-valor {'<' if adf_pvalue < 0.05 else '>='} 0.05: "
          f"{'Rechazar' if adf_pvalue < 0.05 else 'No rechazar'} hipótesis nula - "
          f"{name} es {adf_interpret.lower()}")

    # Prueba KPSS
    kpss_result = kpss(series, regression='c')
    kpss_stat = kpss_result[0]
    kpss_pvalue = kpss_result[1]
    kpss_critical = kpss_result[3]
    kpss_interpret = "No estacionaria" if kpss_pvalue < 0.05 else "Estacionaria"
    kpss_results.append({
        'Serie': name,
        'Estadístico KPSS': f"{kpss_stat:.4f}",
        'p-valor KPSS': f"{kpss_pvalue:.4f}",
        'Valor Crítico 5% KPSS': f"{kpss_critical['5%']:.4f}",
        'Interpretación KPSS': kpss_interpret
    })
    print("\nPrueba KPSS:")
    print(f'Estadístico KPSS: {kpss_stat:.4f}')
    print(f'p-valor: {kpss_pvalue:.4f}')
    print(f'Valores Críticos: {kpss_critical}')
    print(f"Interpretación: p-valor {'<' if kpss_pvalue < 0.05 else '>='} 0.05: "
          f"{'Rechazar' if kpss_pvalue < 0.05 else 'No rechazar'} hipótesis nula - "
          f"{name} es {kpss_interpret.lower()}")

# Realizar pruebas de estacionariedad
unit_root_tests(ipd, "IPD")

# 5. Generar paseo aleatorio
def generate_random_walk(series, steps=12):
    diffs = series.diff().dropna()
    std_dev = diffs.std()
    last_value = series.iloc[-1]
    random_steps = np.random.normal(loc=0, scale=std_dev, size=steps)
    random_walk = [last_value]
    for step in random_steps:
        random_walk.append(random_walk[-1] + step)
    return pd.Series(random_walk[1:], index=pd.date_range(start=series.index[-1] + pd.offsets.MonthBegin(1), periods=steps, freq='M'))

ipd_random_walk = generate_random_walk(ipd, steps=12)

# 6. Encontrar el mejor modelo ARIMA
def find_best_arima(series, name, max_p=3, max_d=2, max_q=3):
    best_aic = float('inf')
    best_order = None
    for p in range(max_p + 1):
        for d in range(max_d + 1):
            for q in range(max_q + 1):
                try:
                    model = ARIMA(series, order=(p, d, q))
                    results = model.fit()
                    if results.aic < best_aic:
                        best_aic = results.aic
                        best_order = (p, d, q)
                except:
                    continue
    print(f"\nMejor modelo ARIMA para {name}:")
    print(f"Orden: {best_order}")
    print(f"AIC: {best_aic:.2f}")
    print("Interpretación:")
    print(f"  - p={best_order[0]}: {best_order[0]} término(s) autorregresivo(s)")
    print(f"  - d={best_order[1]}: {best_order[1]} diferenciación(es) necesaria(s) para estacionariedad")
    print(f"  - q={best_order[2]}: {best_order[2]} término(s) de media móvil")
    return best_order

# Ajustar ARIMA
ipd_order = find_best_arima(ipd, "IPD")
ipd_model = ARIMA(ipd, order=ipd_order).fit()

# 7. Modelos ARCH y GARCH para los retornos
returns = ipd.diff().dropna() * 100  # Retornos como diferencias de logaritmos
def fit_arch_garch(returns, name):
    print(f"\nAnálisis ARCH y GARCH para {name} Retornos:")
    # ARCH(1)
    arch_model_fit = arch_model(returns, vol='ARCH', p=1, rescale=True)
    arch_results = arch_model_fit.fit(disp='off')
    print("\nResumen del modelo ARCH(1):")
    print(arch_results.summary())

    # GARCH(1,1)
    garch_model_fit = arch_model(returns, vol='GARCH', p=1, q=1, rescale=True)
    garch_results = garch_model_fit.fit(disp='off')
    print("\nResumen del modelo GARCH(1,1):")
    print(arch_results.summary())

    # Graficar volatilidad condicional
    plt.figure(figsize=(10, 6))
    plt.plot(arch_results.conditional_volatility, label=f'{name} ARCH(1) Volatilidad')
    plt.plot(garch_results.conditional_volatility, label=f'{name} GARCH(1,1) Volatilidad')
    plt.title(f'Volatilidad Condicional de {name} (ARCH y GARCH)')
    plt.xlabel('Fecha')
    plt.ylabel('Volatilidad')
    plt.legend()
    plt.show()

fit_arch_garch(returns, "IPD")

# 10. Correlograma
def plot_correlograma(series, name, lags=40):
    plt.figure(figsize=(12, 4))
    plt.subplot(1, 2, 1)
    plot_acf(series, lags=lags, ax=plt.gca())
    plt.title(f'{name} ACF')
    plt.subplot(1, 2, 2)
    plot_pacf(series, lags=lags, ax=plt.gca())
    plt.title(f'{name} PACF')
    plt.tight_layout()
    plt.show()

print("\nCorrelograma (Gráficos ACF y PACF):")
